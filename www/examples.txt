# Sample programs for the Reduction Stepper.
#
# Format: a line beginning with "#" starts a new example and gives its name;
# everything up to the next such line is the program. Two things read this file:
# `www/index.html` fetches it to fill the dropdown, and
# `stepper-core/tests/examples.rs` steps every one of them to a value, so an
# example that stops parsing, typechecking or terminating fails the test suite.

# Substitution and tuples
val (x, z) = (5 + 5, 5 - 4)
val y = let val x = 5 in x * z end
val z = x + y

# Short-circuiting
val safe = false andalso 1 < 2
val quick = true orelse 1 < 2

# Nested let, for line breaking
val result = let val a = 1 + 2 val b = a * a in let val c = b + a in c * 2 end end

# Lambdas and application
val twice = fn f : int -> int => fn x : int => f (f x)
val inc = fn n : int => n + 1
val a = twice inc 5

# Currying and partial application
fun add (x : int) (y : int) : int = x + y
val inc = add 1
val a = inc 41

# Recursion: factorial
fun fact (0 : int) : int = 1 
  | fact n = n * fact (n - 1)
val x = fact 5

# fun with several clauses
fun fib (0 : int) : int = 1
  | fib 1 = 1
  | fib n = fib (n - 1) + fib (n - 2)
val x = fib 5

# A recursive function worth folding away
fun sumTo (0 : int) : int = 0 
  | sumTo n = n + sumTo (n - 1)
val total = sumTo 6

# Continuation-passing style
fun factCPS (0 : int) (k : int -> int) : int = k 1
  | factCPS x k = factCPS (x - 1) (fn res : int => k (x * res))
val answer = factCPS 4 (fn x : int => x)
